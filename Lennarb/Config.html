def initialize(app = nil, silent: !ENV["LENNARB_SILENT_LOGS"].nil?, **options)
  self.stderr_output = if silent
    nil
  else
    $stderr
  end

  @app = app
  block = proc { true }
  super(**options, &block)
  apply_defaults_settings
end
